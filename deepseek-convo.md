OBJKT API: Fetching Random Active Listings
The OBJKT API is a GraphQL endpoint at:

https://api.objkt.com/v1/graphql
There’s no built-in “random” operator in GraphQL, so we need a workaround. The typical approach is:

Fetch the total number of active listings

Pick a random offset within that total

Fetch a page of listings at that offset

The core challenge is picking 5 random active listings without fetching all of them. Two strategies:

Strategy A: Random Offset (if total < ~10,000)
Query total active listings.

Choose a random number offset between 0 and total - 5.

Fetch 5 listings at that offset.

Shuffle the 5 and return them.

Pros: Simple.
Cons: OBJKT API may have limits on deep offsets; performance degrades for large totals.

Strategy B: Cached Listing ID Pool (recommended)
Periodically (every 10–30 minutes) fetch a large set of active listing IDs and store them in Redis.

On pack request, randomly pick 5 IDs from the cached pool.

Fetch full details for those IDs using Query 3.

Pros: Fast, scalable, no offset issues.
Cons: Slight delay in new listings appearing in packs (acceptable).

Implementation for Strategy B (simplified):

typescript
// lib/randomPack.ts
import { redis } from './redis';
import { fetchListingsByIds, fetchAllActiveListingIds } from './objkt';

export async function getRandomPack(): Promise<NFTCard[]> {
// Try to get cached IDs; if empty, fetch and cache
let ids: number[] = await redis.smembers('active_listing_ids');
if (ids.length === 0) {
ids = await fetchAllActiveListingIds(); // custom function to paginate through all IDs
await redis.sadd('active_listing_ids', ...ids);
await redis.expire('active_listing_ids', 1800); // 30 min TTL
}

// Pick 5 random unique IDs
const selected = [];
const copy = [...ids];
for (let i = 0; i < 5 && copy.length > 0; i++) {
const idx = Math.floor(Math.random() \* copy.length);
selected.push(copy.splice(idx, 1)[0]);
}

// Fetch full listing details
return await fetchListingsByIds(selected);

tsx
"use client";

import React, { createContext, useContext, useEffect, useState } from "react";
import { BeaconWallet } from "@taquito/beacon-wallet";
import { TezosToolkit } from "@taquito/taquito";
import { NetworkType } from "@airgap/beacon-sdk";

interface WalletContextType {
address: string | null;
connect: () => Promise<void>;
disconnect: () => Promise<void>;
tezos: TezosToolkit | null;
}

const WalletContext = createContext<WalletContextType>({
address: null,
connect: async () => {},
disconnect: async () => {},
tezos: null,
});

export const useWallet = () => useContext(WalletContext);

export const WalletProvider: React.FC<{ children: React.ReactNode }> = ({
children,
}) => {
const [address, setAddress] = useState<string | null>(null);
const [wallet, setWallet] = useState<BeaconWallet | null>(null);
const [tezos, setTezos] = useState<TezosToolkit | null>(null);

useEffect(() => {
// Initialize TezosToolkit and BeaconWallet on mount
const rpcUrl = process.env.NEXT_PUBLIC_TEZOS_RPC_URL!;
const tezosInstance = new TezosToolkit(rpcUrl);
const walletInstance = new BeaconWallet({
name: "TzDeck",
preferredNetwork: NetworkType.MAINNET,
});
tezosInstance.setWalletProvider(walletInstance);
setWallet(walletInstance);
setTezos(tezosInstance);

    // Check if there's an active account (e.g., after page refresh)
    const checkActiveAccount = async () => {
      const activeAccount = await walletInstance.client.getActiveAccount();
      if (activeAccount) {
        setAddress(activeAccount.address);
      }
    };
    checkActiveAccount();

    return () => {
      // Cleanup if needed
    };

}, []);

const connect = async () => {
if (!wallet) return;
try {
await wallet.requestPermissions({
network: { type: NetworkType.MAINNET },
});
const activeAccount = await wallet.client.getActiveAccount();
if (activeAccount) {
setAddress(activeAccount.address);
}
} catch (error) {
console.error("Failed to connect wallet:", error);
}
};

const disconnect = async () => {
if (!wallet) return;
await wallet.clearActiveAccount();
setAddress(null);
};

return (
<WalletContext.Provider value={{ address, connect, disconnect, tezos }}>
{children}
</WalletContext.Provider>
);
};
Explanation:

TezosToolkit is the main Taquito object that talks to the Tezos blockchain.

BeaconWallet is the wallet interface that manages user accounts.

requestPermissions opens the Beacon popup to connect a wallet (Temple, Kukai, etc.).

The useEffect checks for an existing active account on page load, so users stay connected after refresh.

"use client";

import React, { createContext, useContext, useEffect, useState } from "react";
import { BeaconWallet } from "@taquito/beacon-wallet";
import { TezosToolkit } from "@taquito/taquito";
import { NetworkType } from "@airgap/beacon-sdk";

interface WalletContextType {
address: string | null;
connect: () => Promise<void>;
disconnect: () => Promise<void>;
tezos: TezosToolkit | null;
}

const WalletContext = createContext<WalletContextType>({
address: null,
connect: async () => {},
disconnect: async () => {},
tezos: null,
});

export const useWallet = () => useContext(WalletContext);

export const WalletProvider: React.FC<{ children: React.ReactNode }> = ({
children,
}) => {
const [address, setAddress] = useState<string | null>(null);
const [wallet, setWallet] = useState<BeaconWallet | null>(null);
const [tezos, setTezos] = useState<TezosToolkit | null>(null);

useEffect(() => {
// Initialize TezosToolkit and BeaconWallet on mount
const rpcUrl = process.env.NEXT_PUBLIC_TEZOS_RPC_URL!;
const tezosInstance = new TezosToolkit(rpcUrl);
const walletInstance = new BeaconWallet({
name: "TzDeck",
preferredNetwork: NetworkType.MAINNET,
});
tezosInstance.setWalletProvider(walletInstance);
setWallet(walletInstance);
setTezos(tezosInstance);

    // Check if there's an active account (e.g., after page refresh)
    const checkActiveAccount = async () => {
      const activeAccount = await walletInstance.client.getActiveAccount();
      if (activeAccount) {
        setAddress(activeAccount.address);
      }
    };
    checkActiveAccount();

    return () => {
      // Cleanup if needed
    };

}, []);

const connect = async () => {
if (!wallet) return;
try {
await wallet.requestPermissions({
network: { type: NetworkType.MAINNET },
});
const activeAccount = await wallet.client.getActiveAccount();
if (activeAccount) {
setAddress(activeAccount.address);
}
} catch (error) {
console.error("Failed to connect wallet:", error);
}
};

const disconnect = async () => {
if (!wallet) return;
await wallet.clearActiveAccount();
setAddress(null);
};

return (
<WalletContext.Provider value={{ address, connect, disconnect, tezos }}>
{children}
</WalletContext.Provider>
);
};

ts
function convertIpfsUrl(uri: string): string {
if (!uri) return "";

// Handle double ipfs://ipfs/ prefix
if (uri.startsWith("ipfs://ipfs/")) {
return uri.replace("ipfs://ipfs/", "https://ipfs.io/ipfs/");
}

// Handle regular ipfs://
if (uri.startsWith("ipfs://")) {
return uri.replace("ipfs://", "https://ipfs.io/ipfs/");
}

// If it's already an HTTPS URL, return as-is
if (uri.startsWith("https://") || uri.startsWith("http://")) {
return uri;
}

return uri;
}
